cmake_minimum_required(VERSION 3.20)
project(NeonApex VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 5.5 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(raylib)
add_executable(neon_apex src/main.cpp)
target_link_libraries(neon_apex PRIVATE raylib)
enable_testing()
add_test(NAME race_simulation COMMAND neon_apex --self-test)
